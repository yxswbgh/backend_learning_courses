01 零值与容器

02 Struct、方法与接口

type User struct {
    Name string
}

func (u *User) Rename(name string) {
    u.Name = name
}

func (u User) Describe() string {
    return u.Name
}

type Describer interface {
    Describe() string
}

var _ Describer = User{}

值接收者拿副本；需要修改时用指针。接口隐式实现，不需要声明 implements。

03 错误与收尾

if err != nil {
    return fmt.Errorf("load %s: %w", path, err)
}

errors.Is(err, fs.ErrNotExist)
errors.As(err, &pathErr)

defer f.Close()

defer func() {
    if r := recover(); r != nil {
        log.Printf("recovered: %v", r)
    }
}()

普通失败用 error；panic 只处理意外。defer 后进先出。

04 Goroutine / Channel

done := make(chan struct{})
go func() {
    defer close(done)
    work()
}()
<-done

buffered := make(chan int, 10)
buffered <- 1
got := <-buffered

v, ok := <-buffered // closed 时 ok=false

无缓冲：发送与接收会合。有缓冲：缓冲满时发送阻塞。只让发送方 close。

05 Select / Sync

select {
case v := <-ch:
    use(v)
case <-ctx.Done():
    return ctx.Err()
default:
    // 没有 ready case 时立即走这里
}

var mu sync.Mutex
mu.Lock()
defer mu.Unlock()

var wg sync.WaitGroup
wg.Add(1)
go func() { defer wg.Done() }()
wg.Wait()

var once sync.Once
once.Do(initOnce)

Add 在启动 goroutine 前调用。多个 ready case 同时存在时随机选择。

06 Context

ctx, cancel := context.WithTimeout(parent, 2*time.Second)
defer cancel()

select {
case <-ctx.Done():
    return ctx.Err()
case result := <-work:
    return result, nil
}

context 作为第一个参数向下传。取消是协作式信号，不会强杀 goroutine。

08 表格测试与 Benchmark

func TestCount(t *testing.T) {
    tests := []struct {
        name string
        in   []string
        want map[string]int
    }{
        {"one", []string{"a.go"}, map[string]int{".go": 1}},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got := count(tt.in)
            if !reflect.DeepEqual(got, tt.want) {
                t.Fatalf("got %v, want %v", got, tt.want)
            }
        })
    }
}

func BenchmarkCount(b *testing.B) {
    in := []string{"a.go", "b.md"}
    for i := 0; i < b.N; i++ {
        _ = count(in)
    }
}

10 题目：filemix 协议

filemix <dir>

TYPE	COUNT
(none)	1
.env	1
.go	3
TOTAL	4
  • 递归；普通文件；包含隐藏项。
  • 扩展名转小写；无扩展名归为 (none)。
  • 数量降序，同数量类型升序。
  • 用户输入或遍历错误：stderr + 非零退出码。
  • 不能用第三方依赖；测试、race、vet 都要通过。

11 遍历与排序骨架

counts := make(map[string]int)
err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
    if err != nil {
        return err
    }
    if d.Type()&os.ModeSymlink != 0 || d.IsDir() {
        return nil
    }
    if !d.Type().IsRegular() {
        return nil
    }

    ext := strings.ToLower(filepath.Ext(path))
    if ext == "" {
        ext = "(none)"
    }
    counts[ext]++
    return nil
})
rows := make([]row, 0, len(counts))
for ext, n := range counts {
    rows = append(rows, row{ext: ext, count: n})
}

sort.Slice(rows, func(i, j int) bool {
    if rows[i].count != rows[j].count {
        return rows[i].count > rows[j].count
    }
    return rows[i].ext < rows[j].ext
})

12 入口与错误边界

func main() {
    if err := run(os.Args[1:]); err != nil {
        fmt.Fprintf(os.Stderr, "filemix: %v\n", err)
        os.Exit(1)
    }
}

func run(args []string) error {
    if len(args) != 1 {
        return fmt.Errorf("usage: filemix <dir>")
    }

    info, err := os.Stat(args[0])
    if err != nil {
        return fmt.Errorf("stat %s: %w", args[0], err)
    }
    if !info.IsDir() {
        return fmt.Errorf("%s is not a directory", args[0])
    }

    return nil
}

os.Exit 会跳过 defer，所以把可测试逻辑放进 run，main 只负责错误边界。
